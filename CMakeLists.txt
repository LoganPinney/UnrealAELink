cmake_minimum_required(VERSION 3.20)
project(UnrealAELink LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "UnrealAELink milestone zero requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(UnrealAELinkIPC STATIC Shared/src/Transport.cpp)
target_include_directories(UnrealAELinkIPC PUBLIC Shared/include)
if(MSVC)
    target_compile_options(UnrealAELinkIPC PRIVATE /W4 /WX /permissive-)
endif()
add_subdirectory(Tools/ReceiverTest)
include(CTest)
if(BUILD_TESTING)
    add_executable(IpcTests Tests/IpcTests.cpp)
    target_link_libraries(IpcTests PRIVATE UnrealAELinkIPC)
    add_test(NAME WindowsIPC COMMAND IpcTests)
    set_tests_properties(WindowsIPC PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endif()
