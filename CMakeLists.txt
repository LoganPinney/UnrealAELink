cmake_minimum_required(VERSION 3.20)
project(UnrealAELink LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "UnrealAELink milestone zero requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(UnrealAELinkIPC STATIC Shared/src/Transport.cpp Shared/src/RequestTransport.cpp)
target_include_directories(UnrealAELinkIPC PUBLIC Shared/include)
add_library(UnrealAELinkGPU STATIC Shared/src/GpuTransport.cpp)
target_include_directories(UnrealAELinkGPU PUBLIC Shared/include)
target_link_libraries(UnrealAELinkGPU PUBLIC UnrealAELinkIPC d3d12 dxgi)
add_library(UnrealAELinkFrameClient STATIC AfterEffectsPlugin/UnrealAELink/FrameClient.cpp)
target_include_directories(UnrealAELinkFrameClient PUBLIC AfterEffectsPlugin/UnrealAELink)
target_link_libraries(UnrealAELinkFrameClient PUBLIC UnrealAELinkGPU)
if(MSVC)
    target_compile_options(UnrealAELinkIPC PRIVATE /W4 /WX /permissive-)
    target_compile_options(UnrealAELinkGPU PRIVATE /W4 /WX /permissive-)
    target_compile_options(UnrealAELinkFrameClient PRIVATE /W4 /WX /permissive-)
endif()
add_subdirectory(Tools/ReceiverTest)
set(AE_SDK_ROOT "" CACHE PATH "Official Adobe After Effects SDK root containing Examples")
if(AE_SDK_ROOT)
    add_subdirectory(AfterEffectsPlugin/UnrealAELink)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(IpcTests Tests/IpcTests.cpp)
    target_link_libraries(IpcTests PRIVATE UnrealAELinkIPC)
    add_test(NAME WindowsIPC COMMAND IpcTests)
    set_tests_properties(WindowsIPC PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    add_executable(GpuTests Tests/GpuTests.cpp)
    target_link_libraries(GpuTests PRIVATE UnrealAELinkFrameClient)
    add_test(NAME WindowsGPU COMMAND GpuTests)
    set_tests_properties(WindowsGPU PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    add_test(NAME WindowsAdobeFrameClient COMMAND GpuTests --client-test)
    set_tests_properties(WindowsAdobeFrameClient PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
    add_executable(RequestTests Tests/RequestTests.cpp)
    target_link_libraries(RequestTests PRIVATE UnrealAELinkFrameClient)
    if(MSVC)
        target_compile_options(RequestTests PRIVATE /W4 /WX /permissive-)
    endif()
    add_test(NAME WindowsRequests COMMAND RequestTests)
    set_tests_properties(WindowsRequests PROPERTIES TIMEOUT 35 RUN_SERIAL TRUE)
    add_test(NAME WindowsDeterministicGPU COMMAND GpuTests --request-test)
    set_tests_properties(WindowsDeterministicGPU PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endif()
